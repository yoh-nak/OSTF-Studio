# Each VST3 wrapper has distinct JUCE metadata. Its JUCE module objects are
# supplied by the common DSP library compiled in the parent directory.
file(GLOB module_units "${JUCE_MODULES_DIR}/juce_*/*.cpp" "${JUCE_MODULES_DIR}/juce_*/*.mm" "${JUCE_MODULES_DIR}/juce_*/*.c")
list(FILTER module_units EXCLUDE REGEX "/juce_audio_plugin_client/")
set_source_files_properties(${module_units} PROPERTIES HEADER_FILE_ONLY TRUE)
set(stock_names "")
foreach(i RANGE 0 ${last_effect})
    foreach(key type name code category)
        string(JSON ${key} GET "${effects}" ${i} ${key})
    endforeach()
    string(REGEX REPLACE "^OSTF " "" plugin_name "${name}")
    set(target "OSTFFx${type}")
    juce_add_plugin(${target}
        COMPANY_NAME "OSTF Studio" BUNDLE_ID "org.ostf.fx.${code}"
        PLUGIN_MANUFACTURER_CODE Ostf PLUGIN_CODE "${code}" PRODUCT_NAME "${name}" PLUGIN_NAME "${plugin_name}"
        FORMATS VST3 VST3_CATEGORIES Fx "${category}"
        VST3_COPY_DIR "${CMAKE_BINARY_DIR}/StudioEffects/$<CONFIG>" COPY_PLUGIN_AFTER_BUILD FALSE)
    if(type STREQUAL "EffectRack")
        target_sources(${target} PRIVATE ../rack/Factory.cpp)
        target_link_libraries(${target} PRIVATE ostf_rack_host juce::juce_audio_processors)
    elseif(type STREQUAL "Analyzer")
        target_sources(${target} PRIVATE WaveAnalyzer.cpp)
    elseif(type STREQUAL "Unmask")
        target_sources(${target} PRIVATE Unmask.cpp)
    elseif(type STREQUAL "Harmonizer")
        target_sources(${target} PRIVATE Harmonizer.cpp)
        target_include_directories(${target} PRIVATE "${PROJECT_SOURCE_DIR}/third_party/signalsmith")
    elseif(type STREQUAL "StemSeparator")
        target_sources(${target} PRIVATE StemSeparator.cpp ../../ui/StemTranscription.cpp)
        target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../..")
    else()
        target_sources(${target} PRIVATE Factory.cpp)
    endif()
    target_compile_definitions(${target} PRIVATE OSTF_EFFECT_TYPE="${type}" JUCE_VST3_CAN_REPLACE_VST2=0 JUCE_USE_CURL=0)
    target_link_libraries(${target} PRIVATE ostf_effect_dsp juce::juce_recommended_config_flags)
    ostf_attach_notices(${target}_VST3)
    juce_enable_copy_plugin_step(${target})
    string(APPEND stock_names "    {\"${name}\", \"${code}\"},\n")
endforeach()
set(instrument_catalog "${CMAKE_CURRENT_SOURCE_DIR}/../instruments/catalog.json")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${instrument_catalog}")
file(READ "${instrument_catalog}" instruments)
string(JSON instrument_count LENGTH "${instruments}")
math(EXPR last_instrument "${instrument_count} - 1")
foreach(i RANGE 0 ${last_instrument})
    foreach(key type name code)
        string(JSON ${key} GET "${instruments}" ${i} ${key})
    endforeach()
    if(type STREQUAL "ScaleEditor")
        string(APPEND instrument_names "    {\"${name}\", \"${code}\"},\n")
        continue()
    endif()
    set(plugin_name "${name}")
    if(NOT type STREQUAL "Singer")
        string(REGEX REPLACE "^OSTF " "" plugin_name "${name}")
    endif()
    set(target "OSTFSynth${type}")
    juce_add_plugin(${target} COMPANY_NAME "OSTF Studio" BUNDLE_ID "org.ostf.instrument.${code}"
        PLUGIN_MANUFACTURER_CODE Ostf PLUGIN_CODE "${code}" PRODUCT_NAME "${name}" PLUGIN_NAME "${plugin_name}"
        IS_SYNTH TRUE NEEDS_MIDI_INPUT TRUE FORMATS VST3 VST3_CATEGORIES Instrument Synth
        VST3_COPY_DIR "${CMAKE_BINARY_DIR}/StudioEffects/$<CONFIG>" COPY_PLUGIN_AFTER_BUILD FALSE)
    if(type STREQUAL "InstrumentRack")
        target_sources(${target} PRIVATE ../rack/Factory.cpp)
        target_link_libraries(${target} PRIVATE ostf_rack_host juce::juce_audio_processors)
    elseif(type STREQUAL "SuperKnobberLead")
        target_sources(${target} PRIVATE ../instruments/knobber/Lead.cpp
            ../instruments/knobber/Editor.cpp ../instruments/knobber/Engine.cpp)
    elseif(type STREQUAL "Singer")
        juce_add_binary_data(OSTFMeiVoice NAMESPACE MeiVoiceData HEADER_NAME MeiVoiceData.h
            SOURCES "${PROJECT_SOURCE_DIR}/assets/voices/mei-cv.zip" "${PROJECT_SOURCE_DIR}/assets/voices/english-phones.bin")
        target_link_libraries(${target} PRIVATE OSTFMeiVoice PUBLIC OSTFEnglishLexicon)
        target_sources(${target} PRIVATE ../instruments/Singer.cpp ../instruments/VoiceBank.cpp)
    else()
        target_sources(${target} PRIVATE ../instruments/Factory.cpp)
    endif()
    target_compile_definitions(${target} PRIVATE OSTF_ENGINE=Engine::${type} JUCE_VST3_CAN_REPLACE_VST2=0 JUCE_USE_CURL=0)
    target_link_libraries(${target} PRIVATE ostf_effect_dsp juce::juce_recommended_config_flags)
    ostf_attach_notices(${target}_VST3)
    juce_enable_copy_plugin_step(${target})
    string(APPEND instrument_names "    {\"${name}\", \"${code}\"},\n")
endforeach()
set(instrument_names "${instrument_names}" PARENT_SCOPE)
